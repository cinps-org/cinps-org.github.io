rules_version = '2'; service cloud.firestore { match /databases/{database}/documents { // ============================================ // AUTHENTICATION & ROLE HELPERS // ============================================ function signedIn() { return request.auth != null; } // Fetches the admin document once for the authenticated user function getAdminData() { return get(/databases/$(database)/documents/admins/$(request.auth.uid)).data; } function isAdminDocExists() { return exists(/databases/$(database)/documents/admins/$(request.auth.uid)); } function isSuperAdmin() { return signedIn() && isAdminDocExists() && getAdminData().active == true && getAdminData().role == "superAdmin"; } function isAdmin() { return signedIn() && isAdminDocExists() && getAdminData().active == true && getAdminData().role == "admin"; } function isAuthorizedAdmin() { return signedIn() && isAdminDocExists() && getAdminData().active == true && (getAdminData().role == "admin" || getAdminData().role == "superAdmin"); } // Helper to check if the caller is the student owning the record function isStudentOwner(studentId) { return signedIn() && request.auth.uid == studentId; } // ============================================ // ADMIN MANAGEMENT // ============================================ match /admins/{uid} { // Super Admins can manage all admin accounts // Regular Admins can read their own profile doc allow read: if isSuperAdmin() || (signedIn() && request.auth.uid == uid); allow create, delete: if isSuperAdmin(); // Prevent super admins from changing their own active state or role by mistake allow update: if isSuperAdmin() && (request.auth.uid != uid || ( request.resource.data.role == resource.data.role && request.resource.data.active == resource.data.active )); } // ============================================ // STUDENTS // ============================================ match /students/{studentId} { // Admins full access; Students can view their own profile allow read: if isAuthorizedAdmin() || isStudentOwner(studentId); allow write: if isAuthorizedAdmin(); } // ============================================ // FEES // ============================================ match /fees/{feeId} { // Admins full access; Students read fees linked to their UID allow read: if isAuthorizedAdmin() || (signedIn() && resource.data.studentId == request.auth.uid); allow create, update: if isAuthorizedAdmin() && request.resource.data.amount >= 0; allow delete: if isAuthorizedAdmin(); } // ============================================ // PAYMENTS // ============================================ match /payments/{paymentId} { // Admins full access; Students read payments linked to their UID allow read: if isAuthorizedAdmin() || (signedIn() && resource.data.studentId == request.auth.uid); allow create: if isAuthorizedAdmin() && request.resource.data.amount > 0; allow update, delete: if isSuperAdmin(); // Restrict payment edits/deletions to Super Admin for audit safety } // ============================================ // REMINDERS // ============================================ match /reminders/{reminderId} { allow read: if isAuthorizedAdmin() || (signedIn() && resource.data.studentId == request.auth.uid); allow write: if isAuthorizedAdmin(); } // ============================================ // CATCH-ALL DEFAULT // ============================================ // Strictly restrict access to unlisted paths to Super Admin only match /{document=**} { allow read, write: if isSuperAdmin(); } } }
